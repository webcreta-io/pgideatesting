        document.addEventListener('DOMContentLoaded', function() {
            // Create background bubbles
            createBubbles();
            
            // Initialize form functionality
            initForms();
            
            // Load saved data if exists
            loadSavedData();
        });
        
        function createBubbles() {
            const bubblesContainer = document.getElementById('bubbles');
            const bubbleCount = 15;
            
            for (let i = 0; i < bubbleCount; i++) {
                const bubble = document.createElement('div');
                bubble.className = 'bubble';
                
                // Random size between 100px and 300px
                const size = Math.random() * 200 + 100;
                bubble.style.width = `${size}px`;
                bubble.style.height = `${size}px`;
                
                // Random position
                bubble.style.left = `${Math.random() * 100}%`;
                bubble.style.top = `${Math.random() * 100}%`;
                
                // Random opacity
                bubble.style.opacity = Math.random() * 0.3 + 0.1;
                
                bubblesContainer.appendChild(bubble);
            }
        }
        
        function initForms() {
            // Form type selector
            const typeOptions = document.querySelectorAll('.type-option');
            typeOptions.forEach(option => {
                option.addEventListener('click', function() {
                    typeOptions.forEach(opt => opt.classList.remove('selected'));
                    this.classList.add('selected');
                    
                    const formType = this.getAttribute('data-type');
                    document.getElementById('pg-form').style.display = formType === 'pg' ? 'block' : 'none';
                    document.getElementById('mess-form').style.display = formType === 'mess' ? 'block' : 'none';
                    
                    // Reset progress when switching forms
                    updateProgressBar();
                    saveFormState();
                });
            });
            
            // Initialize PG form
            initPgForm();
            
            // Initialize Mess form
            initMessForm();
        }
        
        function initPgForm() {
            const pgSteps = document.querySelectorAll('#pg-form .form-step');
            let currentPgStep = 0;
            
            // Show first step
            showStep(pgSteps, currentPgStep);
            
            // Location selection options
            const locationOptions = document.querySelectorAll('#pg-step1 .option-card');
            locationOptions.forEach(option => {
                option.addEventListener('click', function() {
                    locationOptions.forEach(opt => opt.classList.remove('selected'));
                    this.classList.add('selected');
                    
                    const optionType = this.getAttribute('data-option');
                    document.getElementById('manual-entry-fields').style.display = optionType === 'manual-entry' ? 'block' : 'none';
                    document.getElementById('map-container').style.display = optionType === 'map-pin' ? 'block' : 'none';
                    
                    // Initialize map if selected
                    if (optionType === 'map-pin' && !window.mapInitialized) {
                        initMap();
                        window.mapInitialized = true;
                    }
                    
                    // Get current location if selected
                    if (optionType === 'current-location') {
                        getCurrentLocation();
                    }
                    
                    saveFormState();
                });
            });
            
            // Room charge type toggle
            document.getElementById('pg-charge-entire').addEventListener('change', function() {
                document.getElementById('pg-entire-room-price').style.display = 'block';
                document.getElementById('per-bed-options').style.display = 'none';
                saveFormState();
            });
            
            document.getElementById('pg-charge-per-bed').addEventListener('change', function() {
                document.getElementById('pg-entire-room-price').style.display = 'none';
                document.getElementById('per-bed-options').style.display = 'block';
                saveFormState();
            });
            
            // Beds per room options
            const bedOptions = document.querySelectorAll('input[name="pg-beds"]');
            bedOptions.forEach(option => {
                option.addEventListener('change', function() {
                    document.getElementById('pg-custom-beds').style.display = this.value === 'custom' ? 'block' : 'none';
                    saveFormState();
                });
            });
            
            // Bathroom use conditional
            document.getElementById('pg-bathroom-use').addEventListener('change', function() {
                const sharedOthersContainer = document.getElementById('pg-bathroom-shared-others');
                sharedOthersContainer.classList.toggle('show', this.value === 'shared-others');
                saveFormState();
            });
            
            // Curfew timing toggle
            document.getElementById('pg-curfew-yes').addEventListener('change', function() {
                document.getElementById('pg-curfew-time-container').style.display = 'block';
                saveFormState();
            });
            
            document.getElementById('pg-curfew-no').addEventListener('change', function() {
                document.getElementById('pg-curfew-time-container').style.display = 'none';
                saveFormState();
            });
            
            // Media upload
            initMediaUpload('pg-media-upload', 'pg-media-input', 'pg-media-preview');
            
            // Next button click
            const pgNextButtons = document.querySelectorAll('#pg-form .btn-next');
            pgNextButtons.forEach(button => {
                button.addEventListener('click', function() {
                    if (validateStep(pgSteps[currentPgStep])) {
                        currentPgStep++;
                        showStep(pgSteps, currentPgStep);
                        updateProgressBar();
                        saveFormState();
                    }
                });
            });
            
            // Back button click
            const pgBackButtons = document.querySelectorAll('#pg-form .btn-back');
            pgBackButtons.forEach(button => {
                button.addEventListener('click', function() {
                    currentPgStep--;
                    showStep(pgSteps, currentPgStep);
                    updateProgressBar();
                });
            });
            
            // Submit button
            document.getElementById('pg-submit-btn').addEventListener('click', function() {
                if (validateStep(pgSteps[currentPgStep])) {
                    submitForm('pg');
                }
            });
        }

        function updateProgressBar() {
            const activeForm = document.querySelector('.type-option.selected').getAttribute('data-type');
            const steps = activeForm === 'pg' ? 
                document.querySelectorAll('#pg-form .form-step') : 
                document.querySelectorAll('#mess-form .form-step');

            const progress = (stepIndex / (steps.length - 1)) * 100;

            document.getElementById('progressText').textContent = `${Math.round(progress)}% Complete`;

        function validateStep(step) {
            let isValid = true;
            
            // Location validation
            if (step.id === 'pg-step1') {
                const selectedOption = step.querySelector('.option-card.selected');
                if (!selectedOption) {
                    showError('Please select a location option');
                    isValid = false;
                } else if (selectedOption.getAttribute('data-option') === 'manual-entry') {
                    const address = document.getElementById('pg-address').value;
                    const area = document.getElementById('pg-area').value;
                    const city = document.getElementById('pg-city').value;
                    const state = document.getElementById('pg-state').value;
                    
                    if (!address || !area || !city || !state) {
                        showError('Please fill all address fields');
                        isValid = false;
                    }
                }
            }
            
            // Room charge validation
            if (step.id === 'pg-step2') {
                if (document.getElementById('pg-charge-entire').checked) {
                    const rent = document.getElementById('pg-monthly-rent').value;
                    if (!rent || isNaN(rent) || rent <= 0) {
                        showError('Please enter a valid monthly rent');
                        isValid = false;
                    }
                } else {
                    const beds = document.querySelector('input[name="pg-beds"]:checked');
                    if (!beds) {
                        showError('Please select number of beds');
                        isValid = false;
                    } else if (beds.value === 'custom') {
                        const customBeds = document.getElementById('pg-beds-custom-value').value;
                        if (!customBeds || isNaN(customBeds) || customBeds <= 0) {
                            showError('Please enter a valid number of beds');
                            isValid = false;
                        }
                    }
                    
                    const pricePerBed = document.getElementById('pg-price-per-bed').value;
                    if (!pricePerBed || isNaN(pricePerBed) || pricePerBed <= 0) {
                        showError('Please enter a valid price per bed');
                        isValid = false;
                    }
                }
            }
            
            // Media upload validation
            if (step.id === 'pg-step8') {
                const previewItems = document.querySelectorAll('#pg-media-preview .preview-item');
                if (previewItems.length < 5) {
                    showError('Please upload at least 5 photos');
                    isValid = false;
                }
            }
            
            // Mess menu validation
            if (step.id === 'mess-step1') {
                const menu = document.getElementById('mess-menu').value;
                if (!menu || menu.length < 20) {
                    showError('Please provide a detailed menu (at least 20 characters)');
                    isValid = false;
                }
            }
            
            // Mess media validation
            if (step.id === 'mess-step7') {
                const previewItems = document.querySelectorAll('#mess-media-preview .preview-item');
                if (previewItems.length < 3) {
                    showError('Please upload at least 3 photos of your mess');
                    isValid = false;
                }
            }
            
            return isValid;
        }

        }
        
        function initMediaUpload(uploadContainerId, inputId, previewContainerId) {
            const uploadContainer = document.getElementById(uploadContainerId);
            const fileInput = document.getElementById(inputId);
            const previewContainer = document.getElementById(previewContainerId);
            
            // Click handler
            uploadContainer.addEventListener('click', () => fileInput.click());
            
            // Drag and drop handlers
            uploadContainer.addEventListener('dragover', (e) => {
                e.preventDefault();
                uploadContainer.style.borderColor = 'var(--primary-yellow)';
                uploadContainer.style.backgroundColor = 'rgba(255, 215, 0, 0.05)';
            });
            
            uploadContainer.addEventListener('dragleave', () => {
                uploadContainer.style.borderColor = 'var(--medium-gray)';
                uploadContainer.style.backgroundColor = '';
            });
            
            uploadContainer.addEventListener('drop', (e) => {
                e.preventDefault();
                uploadContainer.style.borderColor = 'var(--medium-gray)';
                uploadContainer.style.backgroundColor = '';
                
                if (e.dataTransfer.files.length) {
                    fileInput.files = e.dataTransfer.files;
                    handleFiles(fileInput.files, previewContainer);
                }
            });
            
            // File input change handler
            fileInput.addEventListener('change', () => {
                if (fileInput.files.length) {
                    handleFiles(fileInput.files, previewContainer);
                }
            });
        }
        
        function handleFiles(files, previewContainer) {
            previewContainer.innerHTML = '';
            
            Array.from(files).forEach(file => {
                if (!file.type.match('image.*') && !file.type.match('video.*')) return;
                
                const reader = new FileReader();
                reader.onload = (e) => {
                    const previewItem = document.createElement('div');
                    previewItem.className = 'preview-item';
                    
                    if (file.type.match('image.*')) {
                        const img = document.createElement('img');
                        img.src = e.target.result;
                        previewItem.appendChild(img);
                    } else if (file.type.match('video.*')) {
                        const video = document.createElement('video');
                        video.src = e.target.result;
                        video.controls = true;
                        previewItem.appendChild(video);
                    }
                    
                    const removeBtn = document.createElement('button');
                    removeBtn.className = 'remove-btn';
                    removeBtn.innerHTML = '×';
                    removeBtn.addEventListener('click', (e) => {
                        e.stopPropagation();
                        previewItem.remove();
                        saveFormState();
                    });
                    
                    previewItem.appendChild(removeBtn);
                    previewContainer.appendChild(previewItem);
                };
                
                reader.readAsDataURL(file);
            });
            
            saveFormState();
        }
